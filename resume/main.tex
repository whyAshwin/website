% Overleaf compiles with pdfLaTeX -> newtx (Times clone), exactly as before.
% Local builds use tectonic (XeTeX), which cannot load newtx; TeX Gyre Termes
% is the same font family newtx is built from, so metrics stay identical.
\usepackage{iftex}
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage{newtxmath,newtxtext}
\else
  % tectonic runs XeTeX. TeX Gyre Termes is the family newtx is derived from,
  % so metrics match the Overleaf build. Its ASCII " slot holds a curly glyph,
  % so map " to a straight quote to keep the output byte-identical to pdfLaTeX.
  \usepackage{fontspec}
  \setmainfont{texgyretermes}[
    Extension      = .otf,
    UprightFont    = *-regular,
    BoldFont       = *-bold,
    ItalicFont     = *-italic,
    BoldItalicFont = *-bolditalic,
    % Keep ff/fi/ffi as separate glyphs so PDF text extraction and ATS parsers
    % read "efficiency" rather than "e\uFB03ciency". Overleaf's T1 build already
    % does this; only the XeTeX path needs the hint.
    Ligatures      = NoCommon
  ]
  \usepackage{textcomp}
  \catcode`\"=\active
  \def"{\textquotedbl}
\fi
